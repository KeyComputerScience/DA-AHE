cmake_minimum_required(VERSION 3.20)
project(daahe_core VERSION 0.2.0 LANGUAGES C)
set(DAAHE_MODULE_RANK "3" CACHE STRING "Compiled module rank (3..16); 3 is legacy")
if(NOT DAAHE_MODULE_RANK MATCHES "^[0-9]+$")
    message(FATAL_ERROR "DAAHE_MODULE_RANK must be an integer between 3 and 16")
endif()
if(DAAHE_MODULE_RANK LESS 3 OR DAAHE_MODULE_RANK GREATER 16)
    message(FATAL_ERROR "DAAHE_MODULE_RANK must be between 3 and 16")
endif()
add_library(daahe_core STATIC daahe_core.c)
target_compile_definitions(daahe_core PUBLIC DAAHE_K=${DAAHE_MODULE_RANK})
target_include_directories(daahe_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(daahe_core PUBLIC c_std_99)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(daahe_core PRIVATE -Wall -Wextra -Wconversion -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(daahe_core_test test_core.c)
    target_link_libraries(daahe_core_test PRIVATE daahe_core)
    add_test(NAME portable_core_known_answers COMMAND daahe_core_test)
endif()
